# 数据库表缺失修复指南

## 问题诊断

**错误日志：**
```
Table 'hotel_system.hotels' doesn't exist
Table 'hotel_system.rooms' doesn't exist
```

**原因：** 数据库中缺少必要的表

## 解决方案

### 3. 创建数据库表

**创建 hotels 表：**

```sql
USE hotel_system;

CREATE TABLE IF NOT EXISTS hotels (
    id INT PRIMARY KEY AUTO_INCREMENT,
    hotel_name VARCHAR(255) NOT NULL COMMENT '酒店名称',
    hotel_address VARCHAR(500) COMMENT '酒店地址',
    hotel_phone VARCHAR(20) COMMENT '酒店电话',
    hotel_star TINYINT DEFAULT 3 COMMENT '酒店星级(1-5)',
    logo VARCHAR(500) COMMENT '酒店Logo',
    description TEXT COMMENT '酒店描述',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='酒店信息表';

-- 插入默认数据
INSERT INTO hotels (id, hotel_name, hotel_address, hotel_phone, hotel_star, description) 
VALUES (1, '智联酒店', '北京市朝阳区', '010-12345678', 4, '智慧酒店物联网控制系统');
```

**创建 rooms 表：**

```sql
CREATE TABLE IF NOT EXISTS rooms (
    id INT PRIMARY KEY AUTO_INCREMENT,
    room_number VARCHAR(50) NOT NULL UNIQUE COMMENT '房间号',
    room_type VARCHAR(50) NOT NULL COMMENT '房间类型(单人间/双人间/套房)',
    room_name VARCHAR(255) COMMENT '房间名称',
    room_price DECIMAL(10,2) NOT NULL COMMENT '房间价格',
    room_status ENUM('available','occupied','maintenance','reserved') DEFAULT 'available' COMMENT '房间状态',
    floor TINYINT COMMENT '楼层',
    area DECIMAL(10,2) COMMENT '面积(平方米)',
    bed_type VARCHAR(50) COMMENT '床型(大床/双床)',
    max_guests TINYINT DEFAULT 2 COMMENT '最多入住人数',
    description TEXT COMMENT '房间描述',
    facilities JSON COMMENT '设施(JSON格式)',
    images JSON COMMENT '图片(JSON格式)',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_room_number (room_number),
    INDEX idx_room_status (room_status),
    INDEX idx_room_type (room_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='房间信息表';
```

**创建其他必要表：**

```sql
-- users 表
CREATE TABLE IF NOT EXISTS users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    email VARCHAR(100),
    phone VARCHAR(20),
    role ENUM('admin','reception','guest') DEFAULT 'guest',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- bookings 表
CREATE TABLE IF NOT EXISTS bookings (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    room_id INT NOT NULL,
    check_in_date DATE NOT NULL,
    check_out_date DATE NOT NULL,
    status ENUM('pending','confirmed','cancelled','checked_in','checked_out') DEFAULT 'pending',
    total_price DECIMAL(10,2),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (room_id) REFERENCES rooms(id),
    INDEX idx_user_id (user_id),
    INDEX idx_room_id (room_id),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- payments 表
CREATE TABLE IF NOT EXISTS payments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_method ENUM('cash','alipay','wechat','credit_card') NOT NULL,
    status ENUM('pending','completed','refunded') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(id),
    INDEX idx_booking_id (booking_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- members 表
CREATE TABLE IF NOT EXISTS members (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    membership_level ENUM('normal','silver','gold','platinum') DEFAULT 'normal',
    points INT DEFAULT 0,

    FOREIGN KEY (user_id) REFERENCES users(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- coupons 表
CREATE TABLE IF NOT EXISTS coupons (
    id INT PRIMARY KEY AUTO_INCREMENT,
    code VARCHAR(50) NOT NULL UNIQUE,
    discount DECIMAL(5,2) NOT NULL,
    min_order DECIMAL(10,2) DEFAULT 0,
    valid_from DATE,
    valid_to DATE,
    usage_limit INT DEFAULT 0,
    used_count INT DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_code (code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- reviews 表
CREATE TABLE IF NOT EXISTS reviews (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    room_id INT NOT NULL,
    rating TINYINT NOT NULL,
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (room_id) REFERENCES rooms(id),
    INDEX idx_user_id (user_id),
    INDEX idx_room_id (room_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- maintenance 表
CREATE TABLE IF NOT EXISTS maintenance (
    id INT PRIMARY KEY AUTO_INCREMENT,
    room_id INT NOT NULL,
    description TEXT NOT NULL,
    status ENUM('pending','in_progress','completed') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (room_id) REFERENCES rooms(id),
    INDEX idx_room_id (room_id),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- delivery 表
CREATE TABLE IF NOT EXISTS delivery (
    id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    item_name VARCHAR(255) NOT NULL,
    delivery_status ENUM('pending','in_transit','delivered','cancelled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_booking_id (booking_id),
    INDEX idx_delivery_status (delivery_status)

-- calls 表
CREATE TABLE IF NOT EXISTS calls (
    id INT PRIMARY KEY AUTO_INCREMENT,
    booking_id INT NOT NULL,
    call_type VARCHAR(50) NOT NULL,
    call_status ENUM('pending','in_progress','completed','cancelled') DEFAULT 'pending',

-- guests 表
CREATE TABLE IF NOT EXISTS guests (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT,
    booking_id INT,
    name VARCHAR(100) NOT NULL,
    id_card VARCHAR(50) NOT NULL,

    check_in_time TIMESTAMP,
```

## 常见问题

### Q: 权限不足

```bash
# 授予用户权限
mysql -u root -p -e "GRANT ALL PRIVILEGES ON hotel_system.* TO 'iot_user'@'localhost'; FLUSH PRIVILEGES;"
```

### Q: 表已存在

```sql
-- 使用 IF NOT EXISTS 避免错误
CREATE TABLE IF NOT EXISTS table_name (...);
```
